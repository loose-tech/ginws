# The Action Chaining System

One of the most powerful architectural features of **GinWS** is the **Action Chaining System**. It allows developers to compose complex, multi-step asynchronous business workflows into clean, decoupled, atomic handlers.

Instead of writing monolithic, deeply nested callback structures or maintaining fragile internal function call trees, an action handler performs its unit of work and returns the **string name** of the next action to execute.

## 1. How It Works

### The Handler Signature

In GinWS, every action handler implements the [`Action`](file:///home/lluie/ws/model.go#L113) type signature:

```go
type Action func(m *Message, client *Client) string
```

- If the function returns a **non-empty string** (e.g. `"order:validate"`), the Hub routes execution to that registered action handler.
- If the function returns an **empty string** (`""`), the pipeline completes.

### Symmetrical Architecture: Server & Client Chaining

Both the **Go server Hub** and the **TypeScript client** implement identical action chaining loops with strict cycle protection (`maxDepth = 50`):

```
                   +───────────────────────────────────+
                   |    Inbound Message { Title }      |
                   +─────────────────┬─────────────────+
                                     │
                                     ▼
                   +───────────────────────────────────+
                   |   Step 1: hub.GetAction(Title)    |
                   +─────────────────┬─────────────────+
                                     │ returns "order:validate"
                                     ▼
                   +───────────────────────────────────+
                   |   Step 2: hub.GetAction(...)      |
                   +─────────────────┬─────────────────+
                                     │ returns "order:finalize"
                                     ▼
                   +───────────────────────────────────+
                   |   Step 3: Terminal Handler        |
                   |   - client.Send(reply)            |
                   +─────────────────┬─────────────────+
                                     │ returns ""
                                     ▼
                             [ Chain Complete ]
```

## 2. Server-Side Execution Loop ([`message.go`](file:///home/lluie/ws/message.go#L9-L55))

When an incoming message arrives, the Hub spawns an asynchronous goroutine to execute the pipeline:

```go
func (h *Hub) process(client *Client, msg *Message, ctx context.Context) error {
    currentName := msg.Title
    reschan := make(chan error, 1)

    go func() {
        depth := 0
        const maxDepth = 50

        for {
            // Check context cancellation / timeout
            if ctx.Err() != nil {
                reschan <- ctx.Err()
                return
            }

            // Loop / recursion protection
            if depth >= maxDepth {
                reschan <- ErrMaxChainDepth
                return
            }
            depth++

            // Lookup registered action handler
            action, ok := h.GetAction(currentName)
            if !ok {
                slog.Debug("no action available", "name", currentName)
                reschan <- ErrFailedToProcess
                return
            }

            // Execute handler and retrieve next route name
            nextRoute := action(msg, client)

            if nextRoute == "" {
                reschan <- nil // Completed chain
                return
            }

            currentName = nextRoute
            slog.Info("chaining to next action", "name", currentName)
        }
    }()

    select {
    case <-ctx.Done():
        return ctx.Err()
    case err := <-reschan:
        return err
    }
}
```

## 3. Passing and Mutating State

Chained actions share the exact same `*ws.Message` instance throughout the entire pipeline. Handlers can inspect and mutate:

1. **`m.Data`**: Designed specifically for internal pipeline metadata, intermediary computation results, or authorization tokens that should not pollute the client-facing payload.
2. **`m.Payload`**: The primary data payload. Handlers can enrich, sanitize, or replace it as the message moves through the pipeline.
3. **`m.Reply`**: The correlation identifier generated by the client. It is preserved across all chain hops so the final action can seamlessly reply back to the awaiting client Promise.

### Example: State Passing Across 3 Steps

```go
ws.On("flow:auth", func(m *ws.Message, c *ws.Client) string {
    // Authenticate and attach user metadata in m.Data
    m.Data = map[string]any{"authenticated": true, "role": "admin"}
    return "flow:enrich"
})

ws.On("flow:enrich", func(m *ws.Message, c *ws.Client) string {
    meta := m.Data.(map[string]any)
    if !meta["authenticated"].(bool) {
        return "" // Stop chain if unauthenticated
    }
    // Enrich payload with database query
    m.Payload = map[string]any{
        "query": m.Payload,
        "results": []string{"Item 1", "Item 2"},
    }
    return "flow:respond"
})

ws.On("flow:respond", func(m *ws.Message, c *ws.Client) string {
    // Terminal action: reply to client using m.Reply
    c.Send(ws.Message{
        Title:   "flow:complete",
        Payload: m.Payload,
        Reply:   m.Reply,
    })
    return "" // Conclude pipeline
})
```

## 4. Built-in Safety & Fault Tolerance

### A. Cycle Protection (`maxDepth = 50`)

If two actions inadvertently reference each other in a circular loop (e.g. `A -> B -> A`):

```go
hub.On("routeA", func(m *ws.Message, c *ws.Client) string {
    return "routeB"
})
hub.On("routeB", func(m *ws.Message, c *ws.Client) string {
    return "routeA"
})
```

The pipeline detects `depth >= 50` and immediately halts execution, returning [`ws.ErrMaxChainDepth`](file:///home/lluie/ws/model.go#L18). If the message included a `Reply` ID, the client's `readPump` automatically transmits an error reply frame:

```json
{
  "title": "error",
  "reply": "client-correlation-id",
  "payload": {
    "error": "maximum action chain depth exceeded"
  }
}
```

The awaiting TypeScript `await ws.request()` call immediately rejects with an `Error("maximum action chain depth exceeded")`.

### B. Global Timeout Context (`DurationMs`)

When a client attaches a timeout duration (e.g. `{ timeout: 3000 }`), the client transmits `duration: 3000`. In Go, the Hub wraps the entire action chain in a `context.WithTimeout`:

```go
// ws/message.go
func (h *Hub) ProcessMessage(client *Client, message *Message) error {
    ctx := context.Background()
    var cancel context.CancelFunc = func() {}

    if message.DurationMs > 0 {
        ctx, cancel = context.WithTimeout(context.Background(), message.DurationMs*time.Millisecond)
    }
    defer cancel()

    return h.process(client, message, ctx)
}
```

If the aggregate execution time across all chained steps exceeds the deadline, the context cancels, interrupting the chain and returning `context.DeadlineExceeded`.

### C. Unknown Action Routing

If a handler returns a route name that is not registered in the Hub:
1. `h.GetAction(currentName)` returns `ok == false`.
2. The pipeline halts with [`ws.ErrFailedToProcess`](file:///home/lluie/ws/model.go#L14).
3. If `m.Reply` is set, an error message is returned to the client and the Promise rejects.

## 5. Real-World Case Studies from the Demo

### Case Study 1: E-Commerce Order Fulfillment (`demo/main.go`)

This 3-stage pipeline ingests an order, validates stock/credentials, and finalizes the transaction:

```go
// Step 1: Ingest order and attach timestamp
ws.On("order:create", func(m *ws.Message, client *ws.Client) string {
    println("[Chain 1/3] order:create processing...")
    m.Payload = map[string]any{
        "item":      m.Payload,
        "step":      "order:create",
        "createdAt": time.Now().Format(time.RFC3339),
    }
    return "order:validate" // Chain to Step 2
})

// Step 2: Validate inventory & authorization
ws.On("order:validate", func(m *ws.Message, client *ws.Client) string {
    println("[Chain 2/3] order:validate verifying stock & auth...")
    data, _ := m.Payload.(map[string]any)
    data["step"] = "order:validate"
    data["validated"] = true
    m.Payload = data
    return "order:finalize" // Chain to Step 3
})

// Step 3: Assign ID, confirm, and reply to client
ws.On("order:finalize", func(m *ws.Message, client *ws.Client) string {
    println("[Chain 3/3] order:finalize completing and replying...")
    data, _ := m.Payload.(map[string]any)
    data["step"] = "order:finalize"
    data["status"] = "confirmed"
    data["orderId"] = "ORD-7749"

    // Echo m.Reply so the client's await ws.request() resolves
    client.Send(ws.Message{
        Title:   "order:completed",
        Payload: data,
        Reply:   m.Reply,
    })
    return "" // Pipeline complete
})
```

### Case Study 2: Multiplayer Tic-Tac-Toe Game Loop (`demo/main.go`)

The interactive demo powers a real-time multiplayer game using a chained game loop:

```
[Client emits "game:move"]
           │
           ▼
[Step 1: "game:move"] ──► Returns "game:validate"
           │
           ▼
[Step 2: "game:validate"]
  - Acquires gameMu.Lock()
  - Validates board slot index (0-8) and vacancy
  - Records player mark ("X" or "O")
  - Alternates turn
  - Returns "game:evaluate"
           │
           ▼
[Step 3: "game:evaluate"]
  - Evaluates 8 winning lines (3 rows, 3 cols, 2 diagonals)
  - Detects draw condition
  - Broadcasts "game:updated" to ALL players
  - If next turn is "O", triggers automated AI bot move
  - Returns ""
```

```go
// Step 1: Ingest move
ws.On("game:move", func(m *ws.Message, client *ws.Client) string {
    return "game:validate"
})

// Step 2: Validate turn, board bounds, and vacancy
ws.On("game:validate", func(m *ws.Message, client *ws.Client) string {
    gameMu.Lock()
    defer gameMu.Unlock()

    moveData, _ := m.Payload.(map[string]any)
    if moveData == nil || game.Winner != "" {
        return ""
    }

    idxFloat, ok := moveData["index"].(float64)
    if !ok {
        return ""
    }
    idx := int(idxFloat)
    if idx < 0 || idx > 8 || game.Board[idx] != "" {
        return ""
    }

    player := game.Turn
    game.Board[idx] = player
    game.LastBy = client.Id()

    // Alternate turn
    if game.Turn == "X" {
        game.Turn = "O"
    } else {
        game.Turn = "X"
    }

    m.Payload = map[string]any{"player": player}
    return "game:evaluate" // Chain to evaluation
})

// Step 3: Evaluate winner or draw, broadcast state, trigger AI opponent
ws.On("game:evaluate", func(m *ws.Message, client *ws.Client) string {
    gameMu.Lock()
    b := game.Board
    lines := [][3]int{
        {0, 1, 2}, {3, 4, 5}, {6, 7, 8}, // rows
        {0, 3, 6}, {1, 4, 7}, {2, 5, 8}, // cols
        {0, 4, 8}, {2, 4, 6},             // diagonals
    }

    for _, line := range lines {
        if b[line[0]] != "" && b[line[0]] == b[line[1]] && b[line[1]] == b[line[2]] {
            game.Winner = b[line[0]]
            if game.Winner == "X" {
                game.ScoreX++
            } else {
                game.ScoreO++
            }
            break
        }
    }

    if game.Winner == "" {
        full := true
        for _, cell := range b {
            if cell == "" {
                full = false
                break
            }
        }
        if full {
            game.Winner = "Draw"
        }
    }

    turnNow := game.Turn
    winnerNow := game.Winner
    gameMu.Unlock()

    // Broadcast updated board to all connected players
    ws.Send(ws.Message{
        Title:   "game:updated",
        Payload: game,
    })

    // If opponent is AI, trigger background move after 600ms
    if winnerNow == "" && turnNow == "O" {
        go triggerAIOpponentMove()
    }

    return ""
})
```

## 6. Client-Side Action Chaining (TypeScript)

The TypeScript [`Ws`](file:///home/lluie/ws/client/ws.ts#L13-L356) client mirrors this chaining capability on the client side via [`ws.use()`](file:///home/lluie/ws/client/ws.ts#L183-L185) and [`ws.runraw()`](file:///home/lluie/ws/client/ws.ts#L283-L324).

```typescript
// client/ws.ts
async runraw(
  map: Map<string, ActionFunc> | Map<string, ReplyHandler>,
  title: string,
  msg: Message,
) {
  let currentTitle: string = title;
  let depth = 0;
  const maxDepth = 50;

  while (currentTitle && depth < maxDepth) {
    depth++;
    const handler = map.get(currentTitle);
    if (!handler) return;

    if (typeof handler === "function") {
      const next = await handler(msg);
      currentTitle = typeof next === "string" ? next : "";
    } else {
      // ReplyHandler resolution...
      return;
    }
  }
}
```

### Example: Client Data Transformation Pipeline

```typescript
// Step 1: Ingest and decrypt/normalize
ws.use('packet:incoming', async (msg: Message) => {
    msg.payload = await decryptPayload(msg.payload);
    return 'packet:validate'; // Chain to client validation
});

// Step 2: Validate schema
ws.use('packet:validate', (msg: Message) => {
    if (!isValid(msg.payload)) {
        console.error('Invalid payload schema');
        return ''; // Abort chain
    }
    return 'packet:render'; // Chain to UI rendering
});

// Step 3: Render to UI
ws.use('packet:render', (msg: Message) => {
    updateComponent(msg.payload);
    return ''; // Conclude client pipeline
});
```
